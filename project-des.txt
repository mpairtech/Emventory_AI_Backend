class VectorStore:
    @staticmethod
    def search(db, embedding):
        sql = text("""
            SELECT product_id,
                   1 - (embedding <=> :q) AS score
            FROM product_vectors
            ORDER BY embedding <=> :q
            LIMIT 10
        """)
        return db.execute(sql, {"q": embedding}).fetchall()from pydantic_settings import BaseSettings

class SearchService:

    @staticmethod
    def index_product(db, payload):
        text = f"{payload['name']} category {payload['category']} price {payload['price']}"
        embedding = EmbeddingService.embed(text)

        vector = ProductVector(
            product_id=payload["product_id"],
            embedding=embedding,
            name=payload["name"],
            category=payload["category"],
            price=payload["price"]
        )
        SearchRepository.upsert(db, vector)

    @staticmethod
    def semantic_search(db, query: str):
        embedding = EmbeddingService.embed(query)
        return VectorStore.search(db, embedding)from fastapi import FastAPI
